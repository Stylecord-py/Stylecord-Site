bot.py · conceptual demonstration 
import stylecord
from stylecord.ext import commands # ext -> extensions

intents = stylecord.Intents()
bot = commands.Bot(intents=intents)

@bot.event
async def on_ready():
    print(f"Bot {bot.user.tag} logged in successfully!")

bot.run("<your token here>")
