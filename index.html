bot.py · demonstração conceitual 
import stylecord
from stylecord.ext import commands # ext -> extensions

intents = stylecord.Intents()
bot = commands.Bot(intents=intents)

@bot.event
async def on_ready():
    print(f"Bot {bot.user.tag} logado com êxito!")

bot.run("<you token here>")
